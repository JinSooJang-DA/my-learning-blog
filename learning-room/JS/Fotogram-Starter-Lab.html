const toggleLike = (id) => {
  setPosts(posts.map(p => {
    if(p.id !== id) return p;
    if(p.liked){
      return {...p, liked:false, likes:p.likes-1}
    } else {
      return {...p, liked:true, likes:p.likes+1}
    }
  }))
}
