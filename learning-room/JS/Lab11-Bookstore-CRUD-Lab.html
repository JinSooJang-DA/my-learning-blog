1. 데이터 초기화
let books = JSON.parse(localStorage.getItem('books'))
            || [...initialBooks];

function saveBooks(){
  localStorage.setItem('books', JSON.stringify(books));
}
문자열↔객체 변환: 저장=stringify, 불러오기=parse
2. Read - render()
function render(){
  contentRef.innerHTML='';
  for(let i=0;i<books.length;i++){
    let book=books[i];
    let commentsHTML='';
    for(let j=0;j<book.comments.length;j++){
      commentsHTML+=`
${book.comments[j].name}
`;
    }
    contentRef.innerHTML+=`
      <div>${book.name}</div>
      ${commentsHTML}
    `;
  }
}
i는 책 번호, j는 댓글 번호
3. Update - toggleLike()
function toggleLike(index){
  books[index].liked = !books[index].liked;
  if(books[index].liked){
    books[index].likes++;
  } else {
    books[index].likes--;
  }
  saveBooks();
  render();
}
!는 반대: true↔false 토글
4. Create - addComment()
function addComment(index){
  let nameRef = document.getElementById(`name-input-${index}`);
  let commentRef = document.getElementById(`comment-input-${index}`);
  let nameVal = nameRef.value.trim();
  let commentVal = commentRef.value.trim();
  if(nameVal===''||commentVal==='') return;
  books[index].comments.push({name:nameVal, comment:commentVal});
  saveBooks(); render();
}
5. Delete - deleteComment()
function deleteComment(bookIndex, commentIndex){
  books[bookIndex].comments.splice(commentIndex,1);
  saveBooks(); render();
}
splice(시작,개수): 1개 삭제
